def _phash(img, hash_size=8):
    size = hash_size * 4
    img = img.convert('L').resize((size, size), Image.LANCZOS)
    pixels = list(img.getdata())
    mean = sum(pixels) / len(pixels)
    return [p > mean for p in pixels]

def _phash_dist(h1, h2):
    return sum(a != b for a, b in zip(h1, h2))

def cl(t):
    if not t: return ""
    t = re.sub(r"t\.me/\S+|http\S+|#[A-Za-z0-9_а-яА-ЯёЁ]+|Источник:.*", "", t, flags=re.I)
    t = re.sub(r'[^\w\s.,!?:;()\-+=%№"\'/]', '', t)
    return " ".join(t.split())


def fetch_tg_link_text(tg_url):
    try:
        embed_url = tg_url.rstrip('/') + '?embed=1&mode=tme'
        r = requests.get(embed_url, timeout=10, headers={
            'User-Agent': 'Mozilla/5.0 (compatible; Googlebot/2.1; +http://www.google.com/bot.html)'
        })
        if r.status_code != 200:
            return ''
        m = re.search(r'class="tgme_widget_message_text[^"]*"[^>]*>(.*?)</div>', r.text, re.DOTALL)
        if m:
            raw = m.group(1)
            raw = re.sub(r'<br\s*/?>', '\n', raw)
            raw = re.sub(r'<[^>]+>', '', raw)
            import html as html_mod
            raw = html_mod.unescape(raw)
            return raw.strip()
    except Exception as ex:
        log.debug(f'fetch_tg_link_text error: {ex}')
    return ''


def enrich_short_with_link(txt):
    link_m = re.search(r'(https?://t\.me/[\w]+/\d+)', txt)
    if not link_m:
        return txt
    text_no_links = re.sub(r'https?://\S+', '', txt).strip()
    if len(text_no_links) < 50:
        fetched = fetch_tg_link_text(link_m.group(1))
        if fetched and len(fetched) >= 15:
            return fetched + '\n\n' + link_m.group(1)
    return txt


def clean_chat_text(t):
    if not t: return ""
    t = EMOJI_RE.sub('', t)
    t = re.sub(r't\.me/\S+|http\S+', '', t)
    return " ".join(t.split()).strip()


def dup(t):
    if not t or len(t) < 20: return False
    c = re.sub(r'[^a-zA-Zа-яА-ЯёЁ0-9]', '', t)
    for o in H:
        if difflib.SequenceMatcher(None, c, o).ratio() > 0.88: return True
    H.append(c)
    if len(H) > 500: H.pop(0)
    return False


def photo_total_size(media):
    try:
        if hasattr(media, 'photo') and hasattr(media.photo, 'sizes'):
            return sum(getattr(s, 'size', 0) for s in media.photo.sizes if hasattr(s, 'size'))
    except Exception:
        pass
    return 0


def dup_by_size_album(photos):
    if len(photos) < 2:
        return False
    total = sum(photo_total_size(m.media) for m in photos)
    if total > 0:
        key = (len(photos), total)
        if key in SENT_ALBUM_SIZES:
            return True
        SENT_ALBUM_SIZES.add(key)
        if len(SENT_ALBUM_SIZES) > MAX_DEDUP_CACHE:
            SENT_ALBUM_SIZES.pop()
    return False


def dup_by_size_single(media):
    sz = photo_total_size(media)
    if sz > 0:
        if sz in SENT_SINGLE_SIZES:
            return True
        SENT_SINGLE_SIZES.add(sz)
        if len(SENT_SINGLE_SIZES) > MAX_DEDUP_CACHE:
            SENT_SINGLE_SIZES.pop()
    return False


async def compute_phash(client, media):
    try:
        buf = io.BytesIO()
        await client.download_media(media, file=buf, thumb=-1)
        buf.seek(0)
        img = Image.open(buf)
        h = _phash(img, hash_size=8)
        buf.close()
        return h
    except Exception as ex:
        log.debug(f'pHash error: {ex}')
        return None


def phash_is_dup(h):
    if h is None:
        return False
    for cached in PHASH_CACHE:
        if _phash_dist(h, cached) <= PHASH_THRESHOLD:
            return True
    PHASH_CACHE.append(h)
    if len(PHASH_CACHE) > MAX_PHASH_CACHE:
        PHASH_CACHE.pop(0)
    return False


async def dup_by_phash_single(client, media):
    h = await compute_phash(client, media)
    if h is not None and phash_is_dup(h):
        return True
    return False


async def dup_by_phash_album(client, photos):
    if not photos:
        return False
    h = await compute_phash(client, photos[0].media)
    return h is not None and phash_is_dup(h)

    async def poll_chat_channels():
        """Polling loop for CHAT channels — works without account membership in group."""

        FETCH_LIMIT   = 8     # messages fetched per channel per cycle
        MIN_LEN       = 5     # min text length to forward
        MAX_LEN       = 300   # max text length (truncate above)

        log.info(f'[CHAT poll] starting — {len(ALL_CHAT_CHANNELS)} channels, every {POLL_INTERVAL}s')
        first_run = True

        while True:
            for n in ALL_CHAT_CHANNELS:
                n_low = n.lower()
                try:
                    entity = await asyncio.wait_for(client.get_entity(n), timeout=12)
                    messages = await asyncio.wait_for(
                        client.get_messages(entity, limit=FETCH_LIMIT), timeout=15
                    )
                    last_known = chat_seen.get(n_low, 0)
                    new_max = last_known
                    for msg in reversed(messages):   # oldest first
                        if msg.id <= last_known:
                            continue
                        new_max = max(new_max, msg.id)
                        if first_run:
                            continue   # on first pass just record ids, don't forward
                        t = clean_chat_text(msg.raw_text or msg.text or '')
                        if not t or len(t) < MIN_LEN:
                            continue
                        if len(t) > MAX_LEN:
                            t = t[:MAX_LEN - 3] + '...'
                        try:
                            src = f"https://t.me/{n_low}/{msg.id}"
                            await client.send_message(
                                DEST['CHAT'], f"{t}\n\n{src}", parse_mode=None
                            )
                            STATS['forwarded'] += 1
                            STATS['last_forward'] = time.strftime('%H:%M:%S UTC', time.gmtime())
                            STATS['per_channel'][n_low] = STATS['per_channel'].get(n_low, 0) + 1
                            log.info(f'[CHAT poll] @{n_low}/{msg.id} -> @{DEST["CHAT"]}')
                            await asyncio.sleep(1)
                        except FloodWaitError as fw:
                            await asyncio.sleep(fw.seconds + 5)
                        except Exception as ex:
                            STATS['errors'] += 1
                            log.warning(f'[CHAT poll send] {n}: {ex}')
                    chat_seen[n_low] = new_max
                except FloodWaitError as fw:
                    await asyncio.sleep(fw.seconds + 10)
                except Exception as ex:
                    log.warning(f'[CHAT poll fetch] {n}: {ex}')
                await asyncio.sleep(BATCH_DELAY)

            if first_run:
                log.info(f'[CHAT poll] first-run baseline done — {len(chat_seen)} channels indexed')
                first_run = False

            await asyncio.sleep(POLL_INTERVAL)

    asyncio.ensure_future(poll_chat_channels())

    entertain_set = {c.lower() for c in entertain_ok}
    med_set = {c.lower() for c in med_ok}

    @client.on(events.NewMessage(chats=all_entertain_ents if all_entertain_ents else ENTERTAIN_CHANNELS))
    async def handle_entertain(e):
        try:
            chat = await e.get_chat()
            un = (getattr(chat, 'username', None) or '').lower()
        except Exception:
            return
        if un not in entertain_set:
            return
        t = clean_chat_text(e.raw_text or e.text or '')
        if not t or len(t) < 3:
            return
        if len(t) > 2000:
            t = t[:1997] + '...'
        try:
            src = f"https://t.me/{un}/{e.id}"
            if e.media and isinstance(e.media, MessageMediaPhoto):
                await client.send_message(DEST['ENTERTAIN'], t or '.', file=e.media, parse_mode=None)
            else:
                await client.send_message(DEST['ENTERTAIN'], f"{t}\n\n{src}", parse_mode=None)
            STATS['forwarded'] += 1
            STATS['last_forward'] = time.strftime('%H:%M:%S UTC', time.gmtime())
            STATS['per_channel'][un] = STATS['per_channel'].get(un, 0) + 1
            log.info(f'[ENTERTAIN] @{un} -> @{DEST["ENTERTAIN"]}')
        except FloodWaitError as fw:
            await asyncio.sleep(fw.seconds + 5)
        except Exception as ex:
            STATS['errors'] += 1

    @client.on(events.NewMessage(chats=all_med_ents if all_med_ents else MED_CHANNELS))
    async def handle_med(e):
        try:
            chat = await e.get_chat()
            un = (getattr(chat, 'username', None) or '').lower()
        except Exception:
            return
        if un not in med_set:
            return
        t = clean_chat_text(e.raw_text or e.text or '')
        if not t or len(t) < 3:
            return
        if len(t) > 2000:
            t = t[:1997] + '...'
        try:
            src = f"https://t.me/{un}/{e.id}"
            if e.media and isinstance(e.media, MessageMediaPhoto):
                await client.send_message(DEST['MED'], t or '.', file=e.media, parse_mode=None)
            else:
                await client.send_message(DEST['MED'], f"{t}\n\n{src}", parse_mode=None)
            STATS['forwarded'] += 1
            STATS['last_forward'] = time.strftime('%H:%M:%S UTC', time.gmtime())
            STATS['per_channel'][un] = STATS['per_channel'].get(un, 0) + 1
            log.info(f'[MED] @{un} -> @{DEST["MED"]}')
        except FloodWaitError as fw:
            await asyncio.sleep(fw.seconds + 5)
        except Exception as ex:
            STATS['errors'] += 1

    total = sum(len(v) for v in STATS['connected'].values())
    log.info(f'Total: {total} channels. Listening...')
    await client.run_until_disconnected()
